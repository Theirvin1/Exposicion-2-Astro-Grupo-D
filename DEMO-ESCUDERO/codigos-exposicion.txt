--- SecurityConfig.java (Segmento de rutas) ---
.authorizeHttpRequests(auth -> auth
    .requestMatchers("/api/auth/**").permitAll()
    .requestMatchers("/api/conductores/**").hasAnyAuthority(
        "ROLE_ADMIN", "ROLE_COORDINADOR"
    )
    .anyRequest().authenticated()
)

--- ConductorController.java ---
@RestController
@RequestMapping("/api/conductores")
public class ConductorController {
    @GetMapping                    // GET /api/conductores
    @GetMapping("/{id}")          // GET /api/conductores/{id}
    @PostMapping                  // POST /api/conductores
    @PutMapping("/{id}")          // PUT /api/conductores/{id}
    @DeleteMapping("/{id}")       // DELETE /api/conductores/{id}
}

--- Conductor Entity (JPA) ---
@Entity
@Table(name = "conductores")
public class Conductor {
    @Id @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;
    private String nombres, apellidos, cedula;
    private String numeroLicencia, tipoLicencia;
    private LocalDate fechaVencimientoLicencia;
    private String telefono, email;
    @Enumerated(EnumType.STRING)
    private EstadoConductor estado;
    private Boolean activo;
    private Instant creadoEn, actualizadoEn;
}

========================================
2. FRONTEND ASTRO (Consumo de APIs)
========================================

--- Login: POST /api/auth/login ---
const res = await fetch(`${API_URL}/api/auth/login`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ email, password })
});
const data = await res.json();  // { accessToken, refreshToken, nombre, email, rol }
localStorage.setItem('token', data.accessToken);

--- Registro: POST /api/auth/register ---
const res = await fetch(`${API_URL}/api/auth/register`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ nombre, email, password, rol })
});

--- Listar Conductores: GET /api/conductores ---
const res = await fetch(`${API_URL}/api/conductores?page=0&size=5`, {
    headers: { 'Authorization': `Bearer ${token}` }
});
const data = await res.json();  // { content: [...], totalPages, totalElements, ... }

--- Crear Conductor: POST /api/conductores ---
const res = await fetch(`${API_URL}/api/conductores`, {
    method: 'POST',
    headers: {
        'Content-Type': 'application/json',
        'Authorization': `Bearer ${token}`
    },
    body: JSON.stringify({
        nombres, apellidos, cedula, numeroLicencia,
        tipoLicencia, fechaVencimientoLicencia,
        telefono, email, estado
    })
});
